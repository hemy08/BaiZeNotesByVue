function lenientParse(str) {
  try { return JSON.parse(str); } catch (_) {}
  var fixed = str
    .replace(/,\s*([\]}])/g, '$1')                       // 去尾随逗号
    .replace(/([{,]\s*)([a-zA-Z_$][a-zA-Z0-9_$]*)(\s*:)/g, '$1"$2"$3'); // 无引号 key 补引号
  return JSON.parse(fixed);
}

8. 时间戳转换器（/front-end/852/）

  • 核心算法：基于 moment 2.x + moment-timezone（带 1970–2030 时区数据）+ zh-cn locale + flatpickr 4.6.13。

9. 进制转换器（/front-end/58/）

function convert() {
  const inputVal = $('#inputNumber').val().trim();
  if (!inputVal) { $('#resultValue').text('--'); return; }
  if (!isValidForBase(inputVal, sourceBase)) throw new Error(`输入包含非 ${sourceBase} 进制字符`);

}
function isValidForBase(val, base) {
  const validChars = "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ".substring(0, base);
  const regex = new RegExp(`^[${validChars}.]+$`, 'i');
  if ((val.match(/\./g) || []).length > 1) return false;
  return regex.test(val);
}

11. 繁简转换（/front-end/5579/）

  • 核心算法：两串等长映射表逐字替换。charPYStr() 返回 4792 字简体串，ftPYStr() 返回对应位置的 4792 字繁体串。traditionalized：逐字符在简体串 indexOf 找位置，命中则取繁体串同位字符，否则原样输出（简→繁）；simplized 反向（繁→简）。无词级转换（如"发"一律映射单个繁体字形，不区分"發/髮"），混合文本无需区分源语言。映射表覆盖 GB2312 一级常用字集规模（4792 字），生僻字不转换。

三、适合作为离线桌面插件实现的优先级排序

  1. Base64 / URL 编码 / HTML 编码 / Unicode / JSON 转义（693、695、691、3602、7683）——全部内联零依赖，几十行即可复刻，无需任何数据表；是最快落地的插件组。
  2. 字数统计（5580）——纯正则，零依赖，即时可用。
  3. 进制转换（58）——仅需 bignumber.js（或原生 BigInt 自实现），支持 2–36 进制+小数，算法简单。
  4. JSON 解析 / 格式化（7438、53）——宽松解析+格式化仅用原生 JSON.parse/stringify；7438 可直接复刻；53 若只保留格式化/压缩/XML/CSV 也可自写（树编辑器才需要 jsoneditor 1 MB）。
  5. 繁简转换（5579）——30 KB 映射表（4792 字对），提取为 JSON 即可，无库依赖。
  6. 汉字转拼音（5523）——125 KB 字库 + 11 KB 工具脚本，离线友好；多音字准确率可用 pinyin-pro 进一步提升。

  8. HTML 转 JavaScript（846）——html2js 逻辑小，js-beautifier 单库即可。
  9. 文本比对（8006）——功能最强但最重（Mergely+CodeMirror ~500 KB、LGPL 许可、Worker 架构）；如降低 UI 要求可换 jsdiff（~15 KB）。
  10. 今日日期/中国日历（5672）——lunar-javascript 344 KB 尚可，但 holidays.json 属外部网络数据且需逐年维护，离线化成本最高，建议最后做或只做农历/干支/节气部分。
