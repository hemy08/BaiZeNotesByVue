```typescript
    EventBus.$off('event-name', handler)
    
    // 2. 清理 IPC 监听器（使用具体函数引用）
    window.electron.ipcRenderer.removeListener('channel', handler)

    if (editorInstance) {

            editorInstance.setModel(null) // 先分离模型

        editorInstance.dispose()

        }
    })
```

#### ✅ 文件缓存策略

**实现位置**: `src/main/utils/baize-store.ts`

**实现方式**:
```typescript
class BaiZeStore {
    private fileCache: Map<string, FileContent> = new Map()
    private readonly MAX_CACHE_SIZE = 50
    
    setFileContent(path: string, content: string): void {
        // LRU 策略：超过最大缓存时删除最旧的
        if (this.fileCache.size >= this.MAX_CACHE_SIZE) {
            const oldestKey = this.fileCache.keys().next().value!
            this.fileCache.delete(oldestKey)
        }
        this.fileCache.set(path, { 
            content, 
            timestamp: Date.now(),
            size: content.length 
        })
    }
}
```

**优点**:
- ✅ LRU 策略防止缓存无限增长
- ✅ 包含时间戳和大小信息
- ✅ 提供统计功能

```typescript
// 性能监控
class PerformanceMonitor {
    private metrics: Map<string, number[]> = new Map()
    
    startMeasure(label: string): string {
        const id = `${label}-${Date.now()}`
        performance.mark(`${id}-start`)
        return id
    }
    
    endMeasure(id: string): number {
        performance.mark(`${id}-end`)
        performance.measure(id, `${id}-start`, `${id}-end`)
        const measure = performance.getEntriesByName(id)[0]
        return measure.duration
    }
    
    getStats(label: string): PerformanceStats {
        const values = this.metrics.get(label) || []
        return {
            avg: sum(values) / values.length,
            max: Math.max(...values),
            min: Math.min(...values)
        }
    }
}
```

```typescript
// src/main/utils/timer-manager.ts
class TimerManager {
    private timers: Map<string, NodeJS.Timeout> = new Map()
    private intervals: Map<string, NodeJS.Timeout> = new Map()
    
    setTimeout(id: string, callback: () => void, delay: number): void {
        if (this.timers.has(id)) {
            clearTimeout(this.timers.get(id)!)
        }
        this.timers.set(id, setTimeout(callback, delay))
    }
    
    setInterval(id: string, callback: () => void, delay: number): void {
        if (this.intervals.has(id)) {
            clearInterval(this.intervals.get(id)!)
        }
        this.intervals.set(id, setInterval(callback, delay))
    }
    
    clearTimeout(id: string): void {
        const timer = this.timers.get(id)
        if (timer) {
            clearTimeout(timer)
            this.timers.delete(id)
        }
    }
    
    clearInterval(id: string): void {
        const interval = this.intervals.get(id)
        if (interval) {
            clearInterval(interval)
            this.intervals.delete(id)
        }
    }
    
    clearAll(): void {
        this.timers.forEach(timer => clearTimeout(timer))
        this.intervals.forEach(interval => clearInterval(interval))
        this.timers.clear()
        this.intervals.clear()
    }
    
    getStats(): TimerStats {
        return {
            timers: this.timers.size,
            intervals: this.intervals.size
        }
    }
}

export const timerManager = new TimerManager()
```

```typescript
// src/main/utils/error-handler.ts
class ErrorHandler {
    static show(error: Error, context?: string) {
        const message = this.formatError(error)
        dialog.showErrorBox('错误', message)
        logger.error(context || 'Error', error)
    }
    
    private static formatError(error: Error): string {
        if (error instanceof FileNotFoundError) {
            return '文件不存在，请检查路径是否正确'
        }
        if (error instanceof PermissionError) {
            return '没有权限访问该文件'
        }
        // ... 更多错误类型
        return error.message
    }
}

// Vue 错误边界
const ErrorBoundary = defineComponent({
    setup(_, { slots }) {
        const error = ref<Error | null>(null)
        
        onErrorCaptured((e) => {
            error.value = e
            return false
        })
        
        return () => {
            if (error.value) {
                return h('div', { class: 'error-boundary' }, [
                    h('h3', '发生错误'),
                    h('p', error.value.message),
                    h('button', { onClick: () => error.value = null }, '重试')
                ])
            }
            return slots.default?.()
        }
    }
})
```
