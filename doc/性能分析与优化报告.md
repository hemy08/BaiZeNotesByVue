## 执行摘要

### 核心问题概述

经过对整个项目的深入分析，发现以下关键性能问题：

#### 🔴 严重问题（需立即解决）

1. **全局变量滥用**: 大量使用 `global` 对象存储状态，导致内存泄漏风险
2. **事件监听器未清理**: EventBus 和 IPC 监听器缺少正确的清理机制
3. **Monaco Editor 实例管理不当**: 编辑器实例未正确销毁，造成内存累积
4. **文件读取无缓存策略**: 每次操作都重新读取文件系统
5. **主题切换全量重渲染**: 主题更新触发整个应用重新渲染

#### 🟡 重要问题（需要优化）

1. **大型组件拆分不足**: 多个组件超过 500 行，职责不清
2. **同步文件操作阻塞**: 主线程执行同步 I/O 操作
3. **Mermaid 渲染阻塞**: 图表渲染在主线程执行
4. **缺少虚拟滚动**: 大文件树渲染性能差
5. **图片处理无压缩**: Base64 图片直接存储，占用大量内存

#### 🟢 改进建议（长期优化）

1. **引入 Web Workers**: 将重型计算移至后台线程
2. **实现增量渲染**: Markdown 预览支持增量更新
3. **数据库替代 JSON**: 使用 SQLite 管理元数据
4. **懒加载优化**: 插件工具按需加载
5. **Bundle 优化**: 减少初始加载体积

#### 1.2 事件总线内存泄漏 🔴

**问题位置**: `src/renderer/src/event-bus.ts`

```typescript
// 当前实现 - 问题代码
const EventBus = {
    callbacks: new Map(),
    
    $on(eventName, callback) {
        if (!this.callbacks.has(eventName)) {
            this.callbacks.set(eventName, [])
        }
        this.callbacks.get(eventName).push(callback)
    },
    
    $emit(eventName, payload?) {
        if (this.callbacks.has(eventName)) {
            this.callbacks.get(eventName).forEach((callback) => {
                callback(payload)
            })
        }
    },
    
    $off(eventName, callback) {
        if (this.callbacks.has(eventName)) {
            this.callbacks.set(
                eventName,
                this.callbacks.get(eventName).filter((cb) => cb !== callback)
            )
        }
    },
    
    $offAll() {
        this.callbacks.clear()
    }
}
```

- ❌ 没有监听器数量限制，可能导致内存爆炸
- ❌ 缺少弱引用支持，回调函数阻止垃圾回收

```typescript
// ✅ 改进的事件总线
interface EventListener {
    callback: Function
    weakRef?: WeakRef<Function>
    once?: boolean
}

class EventBus {
    private callbacks: Map<string, Set<EventListener>> = new Map()
    private readonly MAX_LISTENERS_PER_EVENT = 100
    
    $on(eventName: string, callback: Function, options?: { once?: boolean }): void {
        if (!this.callbacks.has(eventName)) {
            this.callbacks.set(eventName, new Set())
        }
        
        const listeners = this.callbacks.get(eventName)!
        if (listeners.size >= this.MAX_LISTENERS_PER_EVENT) {
            console.warn(`Event "${eventName}" has too many listeners`)
            return
        }
        
        listeners.add({
            callback,
            once: options?.once
        })
    }
    
    $off(eventName: string, callback: Function): void {
        const listeners = this.callbacks.get(eventName)
        if (!listeners) return
        
        for (const listener of listeners) {
            if (listener.callback === callback) {
                listeners.delete(listener)
                break
            }
        }
        
        if (listeners.size === 0) {
            this.callbacks.delete(eventName)
        }
    }
    
    $emit(eventName: string, payload?: any): void {
        const listeners = this.callbacks.get(eventName)
        if (!listeners) return
        
        const toRemove: EventListener[] = []
        
        listeners.forEach(listener => {
            try {
                listener.callback(payload)
                if (listener.once) {
                    toRemove.push(listener)
                }
            } catch (error) {
                console.error(`Error in event listener for "${eventName}":`, error)
            }
        })
        
        toRemove.forEach(listener => listeners.delete(listener))
    }
    
    // 组件卸载时自动清理
    $cleanup(componentId: string): void {
        // 基于组件 ID 清理所有相关监听器
        // 需要在注册时关联组件 ID
    }
}

export default new EventBus()
```

```typescript
// ✅ 使用扁平化索引 + 父子关系
interface FileIndex {
    id: string
    name: string
    path: string
    parentId: string | null
    type: 'file' | 'folder'
    size?: number
    modifiedTime?: number
}

class FileSystemIndex {
    private index: Map<string, FileIndex> = new Map()
    private childrenMap: Map<string, string[]> = new Map()
    
    async buildIndex(rootPath: string): Promise<void> {
        // 使用异步迭代，避免阻塞
        await this.scanDirectory(rootPath, null)
    }
    
    private async scanDirectory(dirPath: string, parentId: string | null): Promise<void> {
        const files = await fs.promises.readdir(dirPath)
        
        for (const file of files) {
            const fullPath = path.join(dirPath, file)
            const stats = await fs.promises.stat(fullPath)
            
            const id = this.generateId(fullPath)
            const entry: FileIndex = {
                id,
                name: file,
                path: fullPath,
                parentId,
                type: stats.isDirectory() ? 'folder' : 'file',
                size: stats.isFile() ? stats.size : undefined,
                modifiedTime: stats.mtimeMs
            }
            
            this.index.set(id, entry)
            
            if (stats.isDirectory()) {
                await this.scanDirectory(fullPath, id)
            }
        }
    }
    
    // 获取指定目录的子节点
    getChildren(parentId: string): FileIndex[] {
        const childrenIds = this.childrenMap.get(parentId) || []
        return childrenIds.map(id => this.index.get(id)!).filter(Boolean)
    }
    
    // 增量更新
    watchForChanges(rootPath: string): void {
        const watcher = chokidar.watch(rootPath, {
            ignored: /node_modules|\.git/,
            persistent: true
        })
        
        watcher
            .on('add', path => this.addToIndex(path))
            .on('unlink', path => this.removeFromIndex(path))
            .on('change', path => this.updateIndex(path))
    }
}
```

```typescript
import { parentPort } from 'worker_threads'

// file-worker.ts - 文件操作工作线程
import * as fs from 'fs/promises'
import * as path from 'path'

parentPort?.on('message', async (task) => {
    try {
        let result
        
        switch (task.type) {
            case 'SAVE_FILE':
                await fs.writeFile(task.path, task.content, 'utf-8')
                result = { success: true }
                break
                
            case 'READ_FILE':
                const content = await fs.readFile(task.path, 'utf-8')
                result = { success: true, content }
                break
                
            case 'DELETE_FILE':
                await fs.unlink(task.path)
                result = { success: true }
                break
                
            case 'RENAME':
                await fs.rename(task.oldPath, task.newPath)
                result = { success: true }
                break
        }
        
        parentPort?.postMessage({ taskId: task.id, result })
    } catch (error) {
        parentPort?.postMessage({
            taskId: task.id,
            error: error.message
        })
    }
})

// 主进程使用
import { Worker } from 'worker_threads'

class FileManager {
    private worker: Worker
    
    constructor() {
        this.worker = new Worker(path.join(__dirname, 'file-worker.js'))
    }
    
    async saveFile(filePath: string, content: string): Promise<void> {
        return new Promise((resolve, reject) => {
            const taskId = Date.now().toString()
            
            const handler = (message: any) => {
                if (message.taskId === taskId) {
                    this.worker.removeListener('message', handler)
                    if (message.error) {
                        reject(new Error(message.error))
                    } else {
                        resolve()
                    }
                }
            }
            
            this.worker.on('message', handler)
            this.worker.postMessage({
                id: taskId,
                type: 'SAVE_FILE',
                path: filePath,
                content
            })
        })
    }
}
```

**改进方案**:

**方案 1: 并行渲染 + 缓存**
```typescript
// ✅ 并行渲染所有图表
async function MermaidRenderAllGraph(text: string): Promise<string> {
    const matches: Array<{ full: string; code: string; index: number }> = []
    const regex = /```mermaid([\s\S]*?)```/g
    let match: RegExpExecArray | null = null
    
    // 第一步：收集所有图表
    while ((match = regex.exec(text)) !== null) {
        matches.push({
            full: match[0],
            code: match[1],
            index: match.index
        })
    }
    
    if (matches.length === 0) return text
    
    // 第二步：并行渲染（限制并发数）
    const concurrencyLimit = 3
    const results = new Map<number, string>()
    
    for (let i = 0; i < matches.length; i += concurrencyLimit) {
        const batch = matches.slice(i, i + concurrencyLimit)
        const promises = batch.map(async (item) => {
            // 检查缓存
            const cached = querySvgByMermaidCode(item.code)
            if (cached) {
                results.set(item.index, cached)
                return
            }
            
            try {
                const svg = await waitAsyncRenderResult(item.code)
                storeMermaidMapping(item.code, svg)
                results.set(item.index, svg)
            } catch (error) {
                console.error('Mermaid render error:', error)
                results.set(item.index, item.code)
            }
        })
        
        await Promise.all(promises)
    }
    
    // 第三步：替换原文（从后往前，避免索引偏移）
    let renderResult = text
    const sortedMatches = [...results.entries()].sort((a, b) => b[0] - a[0])
    
    for (const [index, svg] of sortedMatches) {
        const originalMatch = matches.find(m => m.index === index)!
        const replacement = `<pre class="mermaid"><code>${svg}</code></pre>`
        renderResult = 
            renderResult.substring(0, index) +
            replacement +
            renderResult.substring(index + originalMatch.full.length)
    }
    
    return renderResult
}
```

```typescript
import { diffChars } from 'diff'

class IncrementalRenderer {
    private previousHtml: string = ''
    private cache: Map<string, string> = new Map()
    
    async render(content: string): Promise<string> {
        // 检查缓存
        const cacheKey = this.generateCacheKey(content)
        if (this.cache.has(cacheKey)) {
            return this.cache.get(cacheKey)!
        }
        
        // 对于小改动，尝试增量更新
        if (this.previousHtml && this.isSmallChange(content)) {
            return await this.incrementalUpdate(content)
        }
        
        // 全量渲染
        const html = await this.fullRender(content)
        this.previousHtml = html
        this.cache.set(cacheKey, html)
        
        // 限制缓存大小
        if (this.cache.size > 100) {
            const oldestKey = this.cache.keys().next().value
            this.cache.delete(oldestKey)
        }
        
        return html
    }
    
    private isSmallChange(newContent: string): boolean {
        // 判断是否为小改动（< 10% 内容变化）
        const diff = diffChars(this.previousContent || '', newContent)
        const changedLength = diff
            .filter(d => d.added || d.removed)
            .reduce((sum, d) => sum + (d.value?.length || 0), 0)
        
        return changedLength / newContent.length < 0.1
    }
    
    private async incrementalUpdate(content: string): Promise<string> {
        // 只重新渲染变化的部分
        // 实现较复杂，可使用 virtual-dom 或类似库
        return await this.fullRender(content) // 简化版
    }
}
```

```typescript
// ✅ 图片压缩和优化
import sharp from 'sharp'

async function optimizeImage(buffer: Buffer, options?: {
    maxWidth?: number
    quality?: number
    format?: 'jpeg' | 'png' | 'webp'
}): Promise<Buffer> {
    const {
        maxWidth = 1920,
        quality = 80,
        format = 'webp'
    } = options || {}
    
    let image = sharp(buffer)
    
    // 获取元信息
    const metadata = await image.metadata()
    
    // 如果图片过大，进行缩放
    if (metadata.width && metadata.width > maxWidth) {
        image = image.resize(maxWidth, null, {
            withoutEnlargement: true,
            fit: 'inside'
        })
    }
    
    // 转换为 WebP 格式（更好的压缩率）
    switch (format) {
        case 'webp':
            return await image.webp({ quality }).toBuffer()
        case 'jpeg':
            return await image.jpeg({ quality }).toBuffer()
        case 'png':
            return await image.png({ compressionLevel: 9 }).toBuffer()
    }
}

    if (!imgBuffer) return ''
    
    // 压缩图片

        maxWidth: 1920,
        quality: 80,
        format: 'webp'
    })
    
    // 保存优化后的图片
    const fileName = `${getMathRandom(16)}.webp`
    const outFilePath = path.join(CreateImagesDir(), fileName)
    
    await fs.promises.writeFile(outFilePath, optimizedBuffer)
    
    // 计算压缩率
    const compressionRatio = (1 - optimizedBuffer.length / imgBuffer.length) * 100
    console.log(`Image compressed: ${compressionRatio.toFixed(2)}%`)
    
    return fileName
}
```

```typescript
// performance-monitor.ts
class PerformanceMonitor {
    private metrics: Map<string, MetricData> = new Map()
    private thresholds: Map<string, number> = new Map()
    
    constructor() {
        this.setupThresholds()
        this.startMonitoring()
    }
    
    private setupThresholds() {
        this.thresholds.set('memoryUsage', 500 * 1024 * 1024) // 500 MB
        this.thresholds.set('renderTime', 1000) // 1000 ms
        this.thresholds.set('eventListeners', 1000)
    }
    
    private startMonitoring() {
        // 每 5 秒检查一次
        setInterval(() => {
            this.checkMetrics()
        }, 5000)
    }
    
    recordMetric(name: string, value: number, unit: string = 'ms') {
        const metric = this.metrics.get(name) || {
            values: [],
            min: Infinity,
            max: -Infinity,
            avg: 0
        }
        
        metric.values.push(value)
        if (metric.values.length > 100) {
            metric.values.shift() // 保留最近 100 个值
        }
        
        metric.min = Math.min(metric.min, value)
        metric.max = Math.max(metric.max, value)
        metric.avg = metric.values.reduce((a, b) => a + b, 0) / metric.values.length
        
        this.metrics.set(name, metric)
        
        // 检查阈值
        if (value > (this.thresholds.get(name) || Infinity)) {
            this.alertPerformanceIssue(name, value)
        }
    }
    
    private checkMetrics() {
        const memoryUsage = process.memoryUsage()
        
        this.recordMetric('heapUsed', memoryUsage.heapUsed, 'bytes')
        this.recordMetric('heapTotal', memoryUsage.heapTotal, 'bytes')
        this.recordMetric('rss', memoryUsage.rss, 'bytes')
        
        // 如果内存使用过高，触发 GC
        if (memoryUsage.heapUsed > 400 * 1024 * 1024) {
            if (global.gc) {
                global.gc()
                console.log('Manual GC triggered due to high memory usage')
            }
        }
    }
    
    private alertPerformanceIssue(metric: string, value: number) {
        console.warn(`Performance Alert: ${metric} = ${value} exceeds threshold`)
        
        // 可以发送到监控系统
        // sendToMonitoringService({ metric, value, timestamp: Date.now() })
    }
    
    getReport(): PerformanceReport {
        const report: PerformanceReport = {}
        
        this.metrics.forEach((data, name) => {
            report[name] = {
                current: data.values[data.values.length - 1],
                min: data.min,
                max: data.max,
                avg: data.avg,
                samples: data.values.length
            }
        })
        
        return report
    }
}
```

```javascript
const { session } = require('electron')

class MemoryLeakDetector {
    constructor() {
        this.snapshots = []
        this.interval = null
    }
    
    start(interval = 60000) { // 每分钟
        this.interval = setInterval(() => {
            this.takeSnapshot()
            this.analyze()
        }, interval)
    }
    
    takeSnapshot() {
        const snapshot = {
            timestamp: Date.now(),
            memory: process.memoryUsage(),
            rendererMemory: this.getRendererMemory()
        }
        
        this.snapshots.push(snapshot)
        
        // 保留最近 60 个快照（1 小时）
        if (this.snapshots.length > 60) {
            this.snapshots.shift()
        }
    }
    
    analyze() {
        if (this.snapshots.length < 10) return
        
        const recent = this.snapshots.slice(-10)
        const memoryTrend = this.calculateTrend(recent)
        
        if (memoryTrend.growthRate > 10 * 1024 * 1024) { // 每分钟增长 > 10 MB
            console.warn('Potential memory leak detected!')
            console.warn('Growth rate:', memoryTrend.growthRate / 1024 / 1024, 'MB/min')
            
            // 触发告警
            this.alert(memoryTrend)
        }
    }
    
    calculateTrend(snapshots) {
        const first = snapshots[0].memory.heapUsed
        const last = snapshots[snapshots.length - 1].memory.heapUsed
        const timeDiff = snapshots[snapshots.length - 1].timestamp - snapshots[0].timestamp
        
        return {
            growthRate: (last - first) / (timeDiff / 60000), // MB/min
            totalGrowth: last - first
        }
    }
    
    stop() {
        if (this.interval) {
            clearInterval(this.interval)
        }
    }
}
```
