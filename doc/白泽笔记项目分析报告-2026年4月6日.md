export function TraverseDirectoryLazy(
    dir: string, 
    depth: number = 2, // 限制初始深度
    callback: (fileItems: FileItem[]) => void
) {
    // 只加载前两层
    if (depth <= 0) {

    const cacheKey = `dir-${dir}`

        if (err) return callback([])

        callback(items)
    })
}

#### 5. 事件总线优化

**问题分析**:
- EventBus使用全局事件，容易内存泄漏
- 没有事件清理机制
- 事件监听器重复注册

**优化方案**:

```typescript
// src/renderer/src/event-bus.ts
class OptimizedEventBus {
    private listeners = new Map<string, Set<Function>>()
    
    $on(event: string, callback: Function) {
        if (!this.listeners.has(event)) {
            this.listeners.set(event, new Set())
        }
        this.listeners.get(event).add(callback)
    }
    
    $off(event: string, callback: Function) {
        this.listeners.get(event)?.delete(callback)
        // 自动清理空事件
        if (this.listeners.get(event)?.size === 0) {
            this.listeners.delete(event)
        }
    }
    
    $emit(event: string, ...args: any[]) {
        this.listeners.get(event)?.forEach(cb => cb(...args))
    }
    
    // 清理所有监听器
    $clear() {
        this.listeners.clear()
    }
}
```

**预期效果**: 
- 内存泄漏风险降低

#### 2. 文件内容缓存优化

**问题分析**:
- 打开的文件内容全部缓存在内存
- 没有LRU缓存策略
- 大文件占用过多内存

**优化方案**:

```typescript
// src/main/utils/file-cache.ts
class LRUCache<K, V> {
    private cache = new Map<K, V>()
    private maxSize: number
    
    constructor(maxSize: number = 10) {
        this.maxSize = maxSize
    }
    
    get(key: K): V | undefined {
        const value = this.cache.get(key)
        if (value) {
            // 移到最后（最近使用）
            this.cache.delete(key)
            this.cache.set(key, value)
        }
        return value
    }
    
    set(key: K, value: V) {
        if (this.cache.has(key)) {
            this.cache.delete(key)
        } else if (this.cache.size >= this.maxSize) {
            // 删除最旧的
            const firstKey = this.cache.keys().next().value
            this.cache.delete(firstKey)
        }
        this.cache.set(key, value)
    }
}

const fileCache = new LRUCache<string, string>(20) // 最多缓存20个文件
```

**预期效果**: 
- 内存占用减少50-200MB（取决于文件大小）
